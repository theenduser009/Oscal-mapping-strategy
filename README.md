SELECT
    p.ELEMENT_TYPE      AS PARENT_ELEMENT_TYPE,
    p.NODE_PATH         AS PARENT_NODE_PATH,

    f.DEPENDENCY_TYPE,

    c.ELEMENT_TYPE      AS CHILD_ELEMENT_TYPE,
    c.NODE_PATH         AS CHILD_NODE_PATH,

    f.SEQUENCE

FROM RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.FACT_OSCAL_SSP_DEPENDENCY f

JOIN RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.DIM_OSCAL_SSP_ELEMENTS p
    ON f.FK_SOURCE_ELEMENT_HASH = p.PK_ELEMENT_HASH

JOIN RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.DIM_OSCAL_SSP_ELEMENTS c
    ON f.FK_TARGET_ELEMENT_HASH = c.PK_ELEMENT_HASH

ORDER BY
    p.NODE_PATH,
    f.SEQUENCE,
    c.NODE_PATH;
