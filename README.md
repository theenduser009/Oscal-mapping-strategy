WITH RECURSIVE SYS_IMPL_TREE AS (

    -- Start at system-implementation for one SSP
    SELECT
        0 AS LEVEL_NO,
        d.PK_OSCAL_SSP_ELEMENT_HASH AS NODE_HASH,
        d.ELEMENT_TYPE,
        d.OSCAL_UUID,
        d.SOURCE_RECORD_ID,
        d.METADATA_JSON,
        CAST(d.ELEMENT_TYPE AS VARCHAR) AS TREE_PATH

    FROM RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.DIM_OSCAL_SSP_ELEMENT d

    WHERE d.ELEMENT_TYPE = 'system-implementation'
      AND d.SOURCE_RECORD_ID = '5193'   -- replace with your SSP source record id


    UNION ALL


    -- Follow children downward
    SELECT
        t.LEVEL_NO + 1,
        c.PK_OSCAL_SSP_ELEMENT_HASH,
        c.ELEMENT_TYPE,
        c.OSCAL_UUID,
        c.SOURCE_RECORD_ID,
        c.METADATA_JSON,
        t.TREE_PATH || ' -> ' || c.ELEMENT_TYPE

    FROM SYS_IMPL_TREE t

    JOIN RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.FACT_OSCAL_SSP_DEPENDENCY f
        ON f.FK_SOURCE_ELEMENT_HASH = t.NODE_HASH

    JOIN RTX_ENTERPRISESERVICES_DEV.ES_ESC_GRC_CURATED.DIM_OSCAL_SSP_ELEMENT c
        ON f.FK_TARGET_ELEMENT_HASH = c.PK_OSCAL_SSP_ELEMENT_HASH
)

SELECT
    LEVEL_NO,
    ELEMENT_TYPE,
    TREE_PATH,
    METADATA_JSON
FROM SYS_IMPL_TREE
ORDER BY LEVEL_NO, TREE_PATH;
