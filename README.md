Run this against the raw Archer table to count exactly which IDs are actually present in the three FIPS target groups.

SELECT
    'CONFIDENTIALITY' AS OBJECTIVE,
    f.value::NUMBER AS SELECT_VALUE_ID,
    COUNT(*) AS CNT
FROM RTX_RAW_DEV.ES_ESC_GRC.ARCHER_CONTENT_AUTHORIZATION_PACKAGE_RAW r,
LATERAL FLATTEN(
    INPUT => COALESCE(
        r.CURATED_JSON:CNSS_CONFIDENTIALITY_RATING,
        r.CURATED_JSON:CONFIDENTIALITY_CONTROL_CATEGORY_OVERRIDE,
        r.CURATED_JSON:RECOMMENDED_CONFIDENTIALITY_CONTROL_CATEGORY
    )
) f
GROUP BY 1,2

UNION ALL

SELECT
    'INTEGRITY' AS OBJECTIVE,
    f.value::NUMBER AS SELECT_VALUE_ID,
    COUNT(*) AS CNT
FROM RTX_RAW_DEV.ES_ESC_GRC.ARCHER_CONTENT_AUTHORIZATION_PACKAGE_RAW r,
LATERAL FLATTEN(
    INPUT => COALESCE(
        r.CURATED_JSON:CNSS_INTEGRITY_RATING,
        r.CURATED_JSON:INTEGRITY_CONTROL_CATEGORY_OVERRIDE,
        r.CURATED_JSON:RECOMMENDED_INTEGRITY_CONTROL_CATEGORY,
        r.CURATED_JSON:PROGRAMSITE_INTEGRITY_CONTROL_CATEGORY
    )
) f
GROUP BY 1,2

UNION ALL

SELECT
    'AVAILABILITY' AS OBJECTIVE,
    f.value::NUMBER AS SELECT_VALUE_ID,
    COUNT(*) AS CNT
FROM RTX_RAW_DEV.ES_ESC_GRC.ARCHER_CONTENT_AUTHORIZATION_PACKAGE_RAW r,
LATERAL FLATTEN(
    INPUT => COALESCE(
        r.CURATED_JSON:CNSS_AVAILABILITY_RATING,
        r.CURATED_JSON:AVAILABILITY_CONTROL_CATEGORY_OVERRIDE,
        r.CURATED_JSON:RECOMMENDED_AVAILABILITY_CONTROL_CATEGORY,
        r.CURATED_JSON:PROGRAMSITE_AVAILABILITY_CONTROL_CATEGORY
    )
) f
GROUP BY 1,2

ORDER BY OBJECTIVE, CNT DESC;
